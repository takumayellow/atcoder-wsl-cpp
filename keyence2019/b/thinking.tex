\section*{図: 分け目 $k$ と取り除く区間}
$S$ が \texttt{keyence[:k]} で始まり \texttt{keyence[k:]} で終わるかを $k=0,\dots,7$ で見る（例は $S=$ \texttt{keyofscience}, $k=3$）。

\begin{center}
\begin{tikzpicture}[x=0.62cm,y=0.62cm]
  % 例: S = keyofscience, k = 3
  \foreach \c [count=\i] in {k,e,y,o,f,s,c,i,e,n,c,e} {\node[draw,minimum size=0.6cm,font=\ttfamily] at (\i,0) {\c};}
  \draw[decorate,decoration={brace,amplitude=5pt}] (0.6,0.55) -- (3.4,0.55) node[midway,above=6pt] {\small \texttt{keyence[:3]} で始まる};
  \draw[decorate,decoration={brace,amplitude=5pt}] (8.6,0.55) -- (12.4,0.55) node[midway,above=6pt] {\small \texttt{keyence[3:]} で終わる};
  \draw[redpen,line width=1.2pt] (3.6,-0.55) -- (8.4,-0.55);
  \node[redpen,below] at (6,-0.55) {\small ここ（\texttt{ofsci}）を 1 か所取り除く};
  \node[anchor=west] at (13.2,0) {\small $k=0$ なら先頭、$k=7$ なら末尾を取る};
\end{tikzpicture}
\end{center}
